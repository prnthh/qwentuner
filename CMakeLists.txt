cmake_minimum_required(VERSION 3.20)
project(qwentuner LANGUAGES C CXX)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp)
add_executable(qwentuner-runner native/runner.cpp)
target_link_libraries(qwentuner-runner PRIVATE llama vendor::nlohmann)
target_compile_features(qwentuner-runner PRIVATE cxx_std_17)
set_target_properties(qwentuner-runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
